Unit 5 — Python code

Script used to calculate the Jaccard distances between the three patients. The method and results are written up on the Unit 5 page.

import pandas as pd

# ---------------------------------------------------------
# Data
# ---------------------------------------------------------
data = {
    "Name":   ["Jack", "Mary", "Jim"],
    "Gender": ["M",    "F",    "M"],
    "Fever":  ["Y",    "Y",    "Y"],
    "Cough":  ["N",    "N",    "P"],
    "Test-1": ["P",    "P",    "N"],
    "Test-2": ["N",    "A",    "N"],
    "Test-3": ["N",    "P",    "N"],
    "Test-4": ["A",    "N",    "A"],
}
df = pd.DataFrame(data).set_index("Name")

# ---------------------------------------------------------
# Encode attributes as asymmetric binary variables
#   1 -> "positive" outcome  (Y, P)
#   0 -> "negative" outcome  (N)
#   NaN -> missing / not-applicable (A)
# Gender is a SYMMETRIC attribute -> excluded from Jaccard calc
# ---------------------------------------------------------
def encode(val):
    if val in ("Y", "P"):
        return 1
    elif val == "N":
        return 0
    elif val == "A":
        return None          # treated as missing value
    raise ValueError(f"Unexpected value: {val}")

attr_cols = ["Fever", "Cough", "Test-1", "Test-2", "Test-3", "Test-4"]
bin_df = df[attr_cols].apply(lambda col: col.map(encode))

# ---------------------------------------------------------
# Jaccard distance for asymmetric binary attributes
#   q = # of (1,1) matches, r = # (1,0), s = # (0,1)
#   (0,0) matches are excluded (asymmetric), missing (A) skipped
#   d(i,j) = (r + s) / (q + r + s)
# ---------------------------------------------------------
def jaccard_distance(row_i, row_j):
    q = r = s = 0
    for col in attr_cols:
        vi, vj = row_i[col], row_j[col]
        if vi is None or vj is None:
            continue
        if vi == 1 and vj == 1:
            q += 1
        elif vi == 1 and vj == 0:
            r += 1
        elif vi == 0 and vj == 1:
            s += 1
    denom = q + r + s
    return (r + s) / denom if denom else 0.0

for a, b in [("Jack", "Mary"), ("Jack", "Jim"), ("Jim", "Mary")]:
    print(f"d({a}, {b}) = {jaccard_distance(bin_df.loc[a], bin_df.loc[b]):.3f}")

Unit 6 — Wine K-means code

Script that clusters the Wine chemistry dataset with K-means (K = 3) and compares the clusters with the true wine labels. The method and results are written up on the Unit 6 page.

"""
K-Means Clustering on the Wine Dataset
------------------------------------------
Steps:
1. Load the data.
2. Remove the 'Wine' column before clustering. It's the true class label
  (1, 2, or 3) - not a chemical measurement - and including it would let
  the algorithm "cheat" by clustering directly on the answer. We keep it
  aside separately for evaluation afterward.
3. Standardize the remaining numeric features. This is especially
  important here because the columns are on very different scales
  (e.g., 'Proline' is in the hundreds/thousands while 'Hue' is close to
  1) - without scaling, K-Means' Euclidean distance would be dominated
  by the large-magnitude features.
4. Try several values of K (Elbow Method).
5. Fit K-Means with K = 3 (as required).
6. Compare the 3 clusters to the true 'Wine' labels using a crosstab,
  Adjusted Rand Index (ARI), and Normalized Mutual Information (NMI).
"""

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score
from sklearn.decomposition import PCA

# ------------------------------------------------------------------
# 1. Load data
# ------------------------------------------------------------------
df = pd.read_csv("/mnt/user-data/uploads/Unit06_wine.csv")
print("Original data shape:", df.shape)
print(df.head(), "\n")

# ------------------------------------------------------------------
# 2. Remove the label column ('Wine') before clustering.
#    K-Means uses Euclidean distance over the chemical measurements;
#    the class label itself is not a measurement and must be excluded
#    from the feature matrix (kept aside for later evaluation).
# ------------------------------------------------------------------
X = df.drop(columns=["Wine"])
y_true = df["Wine"]

print("Feature columns used for clustering:", list(X.columns), "\n")

# Standardize features - crucial here since scales vary hugely
# (e.g., Proline ~300-1700 vs Hue ~0.5-1.7)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# ------------------------------------------------------------------
# 3. Elbow method: try K = 1..10
# ------------------------------------------------------------------
inertias = []
K_range = range(1, 11)
for k in K_range:
   km = KMeans(n_clusters=k, random_state=42, n_init=10)
   km.fit(X_scaled)
   inertias.append(km.inertia_)

plt.figure(figsize=(7, 5))
plt.plot(list(K_range), inertias, marker="o")
plt.xlabel("Number of clusters (K)")
plt.ylabel("Inertia (within-cluster sum of squares)")
plt.title("Elbow Method for Optimal K - Wine Dataset")
plt.xticks(list(K_range))
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig("/home/claude/wine_elbow_method.png", dpi=150)
plt.close()
print("Saved elbow plot -> wine_elbow_method.png\n")

# ------------------------------------------------------------------
# 4. Fit K-Means with K = 3 (required)
# ------------------------------------------------------------------
kmeans3 = KMeans(n_clusters=3, random_state=42, n_init=10)
cluster_labels = kmeans3.fit_predict(X_scaled)
df["cluster"] = cluster_labels

print("K = 3 cluster centers (in standardized feature space):")
print(pd.DataFrame(kmeans3.cluster_centers_, columns=X.columns), "\n")

# ------------------------------------------------------------------
# 5. Compare clusters to true Wine labels
# ------------------------------------------------------------------
ct = pd.crosstab(df["Wine"], df["cluster"], rownames=["True Wine Label"], colnames=["Cluster"])
print("Crosstab of true wine class vs. K-Means cluster assignment:")
print(ct, "\n")

ari = adjusted_rand_score(y_true, cluster_labels)
nmi = normalized_mutual_info_score(y_true, cluster_labels)

print(f"Adjusted Rand Index (ARI):        {ari:.4f}")
print(f"Normalized Mutual Information:    {nmi:.4f}")
print("(1.0 = perfect agreement with true wine labels, 0.0 = random)\n")

# Map each cluster to its majority true label, then compute accuracy
cluster_to_label = {}
for cluster_num in sorted(df["cluster"].unique()):
   majority_label = df.loc[df["cluster"] == cluster_num, "Wine"].mode()[0]
   cluster_to_label[cluster_num] = majority_label

df["predicted_wine"] = df["cluster"].map(cluster_to_label)
accuracy = (df["predicted_wine"] == df["Wine"]).mean()
print("Cluster -> majority wine-label mapping:", cluster_to_label)
print(f"Approximate matching accuracy vs. true wine labels: {accuracy:.2%}\n")

# ------------------------------------------------------------------
# Visualize using PCA (reduce 13 chemical features to 2D for plotting)
# ------------------------------------------------------------------
pca = PCA(n_components=2, random_state=42)
X_pca = pca.fit_transform(X_scaled)
df["pca1"] = X_pca[:, 0]
df["pca2"] = X_pca[:, 1]

fig, axes = plt.subplots(1, 2, figsize=(13, 5.5))

label_colors = {1: "tab:blue", 2: "tab:orange", 3: "tab:green"}
for lbl, color in label_colors.items():
   subset = df[df["Wine"] == lbl]
   axes[0].scatter(subset["pca1"], subset["pca2"], label=f"Wine {lbl}", color=color, alpha=0.7)
axes[0].set_title("True Wine Labels (PCA projection)")
axes[0].set_xlabel(f"PC1 ({pca.explained_variance_ratio_[0]:.1%} var)")
axes[0].set_ylabel(f"PC2 ({pca.explained_variance_ratio_[1]:.1%} var)")
axes[0].legend()

scatter = axes[1].scatter(df["pca1"], df["pca2"], c=df["cluster"], cmap="viridis", alpha=0.7)
axes[1].set_title("K-Means Clusters (K = 3, PCA projection)")
axes[1].set_xlabel(f"PC1 ({pca.explained_variance_ratio_[0]:.1%} var)")
axes[1].set_ylabel(f"PC2 ({pca.explained_variance_ratio_[1]:.1%} var)")
legend1 = axes[1].legend(*scatter.legend_elements(), title="Cluster")
axes[1].add_artist(legend1)

plt.tight_layout()
plt.savefig("/home/claude/wine_clusters_vs_labels.png", dpi=150)
plt.close()
print("Saved comparison plot -> wine_clusters_vs_labels.png")

# Save annotated dataframe
df.to_csv("/home/claude/wine_with_clusters.csv", index=False)
print("Saved annotated dataset -> wine_with_clusters.csv")

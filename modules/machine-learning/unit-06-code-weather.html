Unit 6 — Weather (weatherAUS) K-means code

Script that clusters the weatherAUS dataset with K-means for K = 2 to 6, using inertia, silhouette score and PCA plots. The method and results are written up on the Unit 6 page.

"""
K-Means Clustering on the weatherAUS Dataset
------------------------------------------------
Steps:
1. Load the data.
2. Remove columns that are not usable as numeric Euclidean-distance
  features:
    - 'RainToday' and 'RainTomorrow' are categorical (Yes/No) text
      columns, not numeric measurements, so they are dropped from the
      feature matrix used for clustering. ('RainTomorrow' is kept aside
      separately, purely so we can color a plot with it afterward and
      see whether the unsupervised clusters relate to it at all.)
3. Drop rows with missing values in the numeric weather columns (K-Means
  cannot handle NaNs).
4. Standardize the numeric features (they're on very different scales -
  e.g., Pressure ~1000 hPa vs Rainfall ~0-5 mm).
5. Run K-Means for K = 2 through K = 6, plotting inertia (elbow) and
  silhouette scores to compare them.
6. For each K, visualize the clusters with 2D scatter plots using PCA
  (2 principal components) since we have 12 numeric features and can't
  plot them all directly.
"""

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score

# ------------------------------------------------------------------
# 1. Load data
# ------------------------------------------------------------------
df = pd.read_csv("/mnt/user-data/uploads/Unit06_weatherAUS.csv")
print("Original data shape:", df.shape)
print(df.head(), "\n")

# ------------------------------------------------------------------
# 2. Remove non-numeric columns before clustering.
#    RainToday / RainTomorrow are Yes/No text labels, not measurements
#    that a Euclidean distance can be computed on, so they're excluded
#    from the K-Means feature matrix.
# ------------------------------------------------------------------
feature_cols = [c for c in df.columns if c not in ("RainToday", "RainTomorrow")]
X_raw = df[feature_cols].copy()
rain_tomorrow = df["RainTomorrow"].copy()

print("Feature columns used for clustering:", feature_cols, "\n")

# ------------------------------------------------------------------
# 3. Drop rows with missing values in the numeric features
#    (K-Means can't operate on NaNs).
# ------------------------------------------------------------------
mask_complete = X_raw.notna().all(axis=1)
X_raw = X_raw[mask_complete].reset_index(drop=True)
rain_tomorrow = rain_tomorrow[mask_complete].reset_index(drop=True)
print(f"Rows after dropping missing values: {X_raw.shape[0]} (out of {df.shape[0]} original)\n")

# ------------------------------------------------------------------
# 4. Standardize features
# ------------------------------------------------------------------
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_raw)

# For speed on 100k+ rows, use a random subsample for silhouette
# score calculation (silhouette is O(n^2) and too slow on full data)
rng = np.random.default_rng(42)
sample_idx = rng.choice(X_scaled.shape[0], size=min(5000, X_scaled.shape[0]), replace=False)

# ------------------------------------------------------------------
# 5. Run K-Means for K = 2..6, collect inertia + silhouette score
# ------------------------------------------------------------------
K_values = range(2, 7)
inertias = []
silhouettes = []
all_labels = {}

for k in K_values:
   km = KMeans(n_clusters=k, random_state=42, n_init=10)
   labels = km.fit_predict(X_scaled)
   all_labels[k] = labels
   inertias.append(km.inertia_)
   sil = silhouette_score(X_scaled[sample_idx], labels[sample_idx])
   silhouettes.append(sil)
   print(f"K={k}: inertia={km.inertia_:.1f}, silhouette (sampled)={sil:.4f}")

# Plot inertia and silhouette side by side
fig, axes = plt.subplots(1, 2, figsize=(13, 5))
axes[0].plot(list(K_values), inertias, marker="o", color="tab:blue")
axes[0].set_xlabel("Number of clusters (K)")
axes[0].set_ylabel("Inertia")
axes[0].set_title("Elbow Method")
axes[0].set_xticks(list(K_values))
axes[0].grid(True, alpha=0.3)

axes[1].plot(list(K_values), silhouettes, marker="o", color="tab:orange")
axes[1].set_xlabel("Number of clusters (K)")
axes[1].set_ylabel("Silhouette Score (sampled)")
axes[1].set_title("Silhouette Score vs K")
axes[1].set_xticks(list(K_values))
axes[1].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig("/home/claude/weather_elbow_silhouette.png", dpi=150)
plt.close()
print("\nSaved elbow/silhouette plot -> weather_elbow_silhouette.png\n")

# ------------------------------------------------------------------
# 6. 2D visualization via PCA for each K = 2..6
# ------------------------------------------------------------------
pca = PCA(n_components=2, random_state=42)
X_pca = pca.fit_transform(X_scaled)
print(f"PCA explained variance ratio (PC1, PC2): {pca.explained_variance_ratio_}\n")

fig, axes = plt.subplots(2, 3, figsize=(18, 11))
axes = axes.flatten()

for i, k in enumerate(K_values):
   ax = axes[i]
   labels = all_labels[k]
   scatter = ax.scatter(X_pca[:, 0], X_pca[:, 1], c=labels, cmap="viridis",
                         s=4, alpha=0.4)
   ax.set_title(f"K = {k}")
   ax.set_xlabel(f"PC1 ({pca.explained_variance_ratio_[0]:.1%} var)")
   ax.set_ylabel(f"PC2 ({pca.explained_variance_ratio_[1]:.1%} var)")

# Use last subplot slot to show RainTomorrow for reference
ax = axes[5]
colors = rain_tomorrow.map({"Yes": "tab:red", "No": "tab:blue"}).fillna("lightgray")
ax.scatter(X_pca[:, 0], X_pca[:, 1], c=colors.to_numpy(), s=4, alpha=0.3)
ax.set_title("Reference: RainTomorrow (red=Yes, blue=No)")
ax.set_xlabel(f"PC1 ({pca.explained_variance_ratio_[0]:.1%} var)")
ax.set_ylabel(f"PC2 ({pca.explained_variance_ratio_[1]:.1%} var)")

plt.tight_layout()
plt.savefig("/home/claude/weather_clusters_k2to6.png", dpi=150)
plt.close()
print("Saved cluster comparison plot (K=2..6) -> weather_clusters_k2to6.png\n")

# ------------------------------------------------------------------
# Also show 2D scatter using two real (non-PCA) features for intuition:
# Humidity3pm vs MaxTemp, for K = 3 as an illustrative example
# ------------------------------------------------------------------
k_example = 3
labels_example = all_labels[k_example]
fig, ax = plt.subplots(figsize=(8, 6))
scatter = ax.scatter(X_raw["MaxTemp"], X_raw["Humidity3pm"], c=labels_example,
                     cmap="viridis", s=4, alpha=0.4)
ax.set_xlabel("MaxTemp")
ax.set_ylabel("Humidity3pm")
ax.set_title(f"K-Means Clusters (K={k_example}) shown on real features")
legend1 = ax.legend(*scatter.legend_elements(), title="Cluster")
ax.add_artist(legend1)
plt.tight_layout()
plt.savefig("/home/claude/weather_k3_real_features.png", dpi=150)
plt.close()
print("Saved real-feature scatter plot (K=3) -> weather_k3_real_features.png\n")

# ------------------------------------------------------------------
# Save clustered data (K=3 as a representative example) to CSV
# ------------------------------------------------------------------
X_raw["cluster_k3"] = all_labels[3]
X_raw["RainTomorrow"] = rain_tomorrow.values
X_raw.to_csv("/home/claude/weatherAUS_with_clusters.csv", index=False)
print("Saved annotated dataset (with K=3 cluster labels) -> weatherAUS_with_clusters.csv")

# Crosstab: does the cluster relate at all to RainTomorrow?
ct = pd.crosstab(X_raw["RainTomorrow"], X_raw["cluster_k3"])
print("\nCrosstab of RainTomorrow vs K=3 cluster (for reference/interest only):")
print(ct)

Unit 6 — Iris K-means code

Script that clusters the Iris dataset with K-means (K = 3) and compares the clusters with the true species. The method and results are written up on the Unit 6 page.

"""
K-Means Clustering on the Iris Dataset
----------------------------------------
Steps:
1. Load the data.
2. Drop the non-numeric 'species' column (K-Means needs numeric features
  because it computes Euclidean distances - text labels can't be used).
3. Try several values of K using the Elbow Method (inertia) to see how
  many clusters the data naturally suggests.
4. Fit K-Means with K = 3 (as required).
5. Compare the 3 clusters found by K-Means to the true 'species' labels
  using a crosstab/confusion-matrix-style table, Adjusted Rand Index (ARI)
  and Normalized Mutual Information (NMI) - both are standard metrics for
  comparing a clustering result to known ground-truth labels.
"""

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score, confusion_matrix

# ------------------------------------------------------------------
# 1. Load data
# ------------------------------------------------------------------
df = pd.read_csv("/mnt/user-data/uploads/Unit06_iris.csv")
print("Original data shape:", df.shape)
print(df.head(), "\n")

# ------------------------------------------------------------------
# 2. Remove the non-numeric column ('species') before clustering.
#    K-Means relies on Euclidean distance, which is only defined for
#    numeric features, so the species label must be excluded from
#    the feature matrix (we keep it aside for later evaluation).
# ------------------------------------------------------------------
X = df.drop(columns=["species"])
y_true = df["species"]

print("Feature columns used for clustering:", list(X.columns), "\n")

# Standardize features (recommended for K-Means so that no single
# feature with a larger scale dominates the distance calculation)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# ------------------------------------------------------------------
# 3. Elbow method: try K = 1..10 to see how inertia changes
# ------------------------------------------------------------------
inertias = []
K_range = range(1, 11)
for k in K_range:
   km = KMeans(n_clusters=k, random_state=42, n_init=10)
   km.fit(X_scaled)
   inertias.append(km.inertia_)

plt.figure(figsize=(7, 5))
plt.plot(list(K_range), inertias, marker="o")
plt.xlabel("Number of clusters (K)")
plt.ylabel("Inertia (within-cluster sum of squares)")
plt.title("Elbow Method for Optimal K")
plt.xticks(list(K_range))
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig("/home/claude/elbow_method.png", dpi=150)
plt.close()
print("Saved elbow plot -> elbow_method.png\n")

# ------------------------------------------------------------------
# 4. Fit K-Means with K = 3 (required)
# ------------------------------------------------------------------
kmeans3 = KMeans(n_clusters=3, random_state=42, n_init=10)
cluster_labels = kmeans3.fit_predict(X_scaled)
df["cluster"] = cluster_labels

print("K = 3 cluster centers (in standardized feature space):")
print(pd.DataFrame(kmeans3.cluster_centers_, columns=X.columns), "\n")

# ------------------------------------------------------------------
# 5. Compare clusters to true species labels
# ------------------------------------------------------------------
# Crosstab: rows = true species, columns = predicted cluster
ct = pd.crosstab(df["species"], df["cluster"], rownames=["Species"], colnames=["Cluster"])
print("Crosstab of true species vs. K-Means cluster assignment:")
print(ct, "\n")

ari = adjusted_rand_score(y_true, cluster_labels)
nmi = normalized_mutual_info_score(y_true, cluster_labels)

print(f"Adjusted Rand Index (ARI):        {ari:.4f}")
print(f"Normalized Mutual Information:    {nmi:.4f}")
print("(1.0 = perfect agreement with true species labels, 0.0 = random)\n")

# Simple "accuracy": map each cluster to its majority species, then
# check what fraction of points match.
mapping = ct.idxmax(axis=0)  # majority species for each cluster number... actually need cluster->species
# Build cluster -> most common true species
cluster_to_species = {}
for cluster_num in sorted(df["cluster"].unique()):
   majority_species = df.loc[df["cluster"] == cluster_num, "species"].mode()[0]
   cluster_to_species[cluster_num] = majority_species

df["predicted_species"] = df["cluster"].map(cluster_to_species)
accuracy = (df["predicted_species"] == df["species"]).mean()
print("Cluster -> majority species mapping:", cluster_to_species)
print(f"Approximate matching accuracy vs. true species: {accuracy:.2%}\n")

# ------------------------------------------------------------------
# Visualize clusters vs. true species (using petal length/width,
# the two most separating features)
# ------------------------------------------------------------------
fig, axes = plt.subplots(1, 2, figsize=(13, 5.5))

species_colors = {"setosa": "tab:blue", "versicolor": "tab:orange", "virginica": "tab:green"}
for sp, color in species_colors.items():
   subset = df[df["species"] == sp]
   axes[0].scatter(subset["petal_length"], subset["petal_width"], label=sp, color=color, alpha=0.7)
axes[0].set_title("True Species Labels")
axes[0].set_xlabel("Petal Length")
axes[0].set_ylabel("Petal Width")
axes[0].legend()

scatter = axes[1].scatter(df["petal_length"], df["petal_width"], c=df["cluster"], cmap="viridis", alpha=0.7)
axes[1].set_title("K-Means Clusters (K = 3)")
axes[1].set_xlabel("Petal Length")
axes[1].set_ylabel("Petal Width")
legend1 = axes[1].legend(*scatter.legend_elements(), title="Cluster")
axes[1].add_artist(legend1)

plt.tight_layout()
plt.savefig("/home/claude/clusters_vs_species.png", dpi=150)
plt.close()
print("Saved comparison plot -> clusters_vs_species.png")

# Save the annotated dataframe (with cluster + predicted species) as CSV
df.to_csv("/home/claude/iris_with_clusters.csv", index=False)
print("Saved annotated dataset -> iris_with_clusters.csv")

Unit 2 — Python code

← Back to Unit 2

Script used for the Auto-MPG exploratory data analysis. It was run in Google Colab on Unit02_auto-mpg__1_.csv; the results are written up on the Unit 2 page.

# ============================================================
# Exploratory Data Analysis - Auto MPG Dataset
# ============================================================

from google.colab import files
uploaded = files.upload()   # select Unit02_auto-mpg__1_.csv when prompted

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import skew, kurtosis
from scipy.cluster.hierarchy import dendrogram, linkage
from scipy.spatial.distance import squareform
from sklearn.preprocessing import StandardScaler

sns.set(style="whitegrid")

# Load the CSV
df = pd.read_csv("Unit02_auto-mpg__1_.csv")

# Quick look at the data
print("Shape:", df.shape)
df.head()

# ============================================================
# 1. IDENTIFY MISSING VALUES
# ============================================================

# The 'horsepower' column is stored as text and uses '?' for missing values.
# First, convert it to numeric, coercing '?' into NaN.
df['horsepower'] = pd.to_numeric(df['horsepower'], errors='coerce')

print("\n--- Missing values per column (after converting horsepower) ---")
print(df.isnull().sum())

print("\n--- Percentage missing per column ---")
print((df.isnull().mean() * 100).round(2))

# Visualize missing values
plt.figure(figsize=(8, 4))
sns.heatmap(df.isnull(), cbar=False, cmap="viridis", yticklabels=False)
plt.title("Missing Values Heatmap")
plt.tight_layout()
plt.show()

# Handle missing values: impute horsepower with the median
df['horsepower'] = df['horsepower'].fillna(df['horsepower'].median())

print("\nMissing values after imputation:")
print(df.isnull().sum())

# ============================================================
# 2. ESTIMATE SKEWNESS AND KURTOSIS
# ============================================================

numeric_cols = ['mpg', 'cylinders', 'displacement', 'horsepower',
                 'weight', 'acceleration', 'model year']

skew_kurt = pd.DataFrame({
    'Skewness': [skew(df[col]) for col in numeric_cols],
    'Kurtosis': [kurtosis(df[col]) for col in numeric_cols]  # Fisher's definition (normal = 0)
}, index=numeric_cols)

print("\n--- Skewness and Kurtosis of numeric features ---")
print(skew_kurt.round(3))

# Visualize distribution shapes
fig, axes = plt.subplots(3, 3, figsize=(15, 12))
axes = axes.flatten()
for i, col in enumerate(numeric_cols):
    sns.histplot(df[col], kde=True, ax=axes[i], color="steelblue")
    axes[i].set_title(f"{col}\nSkew={skew(df[col]):.2f}, Kurt={kurtosis(df[col]):.2f}")
for j in range(len(numeric_cols), len(axes)):
    fig.delaxes(axes[j])
plt.tight_layout()
plt.show()

# ============================================================
# 3. CORRELATION HEAT MAP
# ============================================================

corr_matrix = df[numeric_cols].corr()

plt.figure(figsize=(9, 7))
sns.heatmap(corr_matrix, annot=True, fmt=".2f", cmap="coolwarm",
            square=True, linewidths=0.5)
plt.title("Correlation Heatmap - Auto MPG Features")
plt.tight_layout()
plt.show()

# ============================================================
# 4. SCATTER PLOTS FOR DIFFERENT PARAMETERS
# ============================================================

# Pairwise scatter plot matrix
sns.pairplot(df[numeric_cols], diag_kind="kde", corner=True,
             plot_kws={'alpha': 0.6, 's': 25, 'edgecolor': 'k'})
plt.suptitle("Pairwise Scatter Plots - Auto MPG Features", y=1.02)
plt.show()

# A few individual scatter plots of specific relationships
fig, axes = plt.subplots(2, 2, figsize=(12, 10))

sns.scatterplot(data=df, x="weight", y="mpg", ax=axes[0, 0], color="teal")
axes[0, 0].set_title("MPG vs Weight")

sns.scatterplot(data=df, x="horsepower", y="mpg", ax=axes[0, 1], color="darkorange")
axes[0, 1].set_title("MPG vs Horsepower")

sns.scatterplot(data=df, x="displacement", y="mpg", ax=axes[1, 0], color="purple")
axes[1, 0].set_title("MPG vs Displacement")

sns.scatterplot(data=df, x="acceleration", y="mpg", ax=axes[1, 1], color="crimson")
axes[1, 1].set_title("MPG vs Acceleration")

plt.tight_layout()
plt.show()

# ============================================================
# 5. DENDROGRAM (Hierarchical Clustering)
# ============================================================

# --- 5a. Feature clustering ---
# Groups variables that behave similarly, using (1 - correlation) as the
# distance metric. Complements the correlation heatmap above.
corr_dist = 1 - corr_matrix.abs()
condensed_dist = squareform(corr_dist.values, checks=False)
feature_linkage = linkage(condensed_dist, method="average")

plt.figure(figsize=(9, 5))
dendrogram(feature_linkage, labels=corr_matrix.columns.tolist(),
           leaf_rotation=90, color_threshold=0.3)
plt.title("Feature Dendrogram (based on correlation distance)")
plt.ylabel("Distance (1 - |correlation|)")
plt.tight_layout()
plt.show()

# --- 5b. Car (observation) clustering ---
# Groups similar cars based on standardized numeric specs.
# Ward linkage minimizes within-cluster variance - good default for this.
scaler = StandardScaler()
scaled_features = scaler.fit_transform(df[numeric_cols])
car_linkage = linkage(scaled_features, method="ward")

plt.figure(figsize=(12, 6))
dendrogram(car_linkage, truncate_mode="lastp", p=30,
           show_leaf_counts=True, leaf_rotation=90)
plt.title("Car Dendrogram (Ward linkage, last 30 merged clusters)")
plt.xlabel("Cluster size (number of cars)")
plt.ylabel("Distance")
plt.tight_layout()
plt.show()

# ============================================================
# 6. REPLACE CATEGORICAL VALUES WITH NUMERICAL VALUES
# ============================================================

# In this dataset, 'origin' is the categorical feature representing the
# country/region of manufacture:
#   1 = America (USA), 2 = Europe, 3 = Japan/Asia
# It is already encoded numerically in the raw file, but let's demonstrate
# the mapping explicitly (useful if you ever get a version with text labels).

origin_map_text_to_num = {"America": 1, "Europe": 2, "Japan": 3, "Asia": 3}
origin_map_num_to_text = {1: "America", 2: "Europe", 3: "Japan"}

# Create a readable label column for reference/EDA purposes
df['origin_label'] = df['origin'].map(origin_map_num_to_text)

# Confirm origin is numeric (int) as required for modeling
df['origin'] = df['origin'].map(origin_map_text_to_num).fillna(df['origin']).astype(int)

print("\n--- Origin encoding check ---")
print(df[['origin', 'origin_label']].drop_duplicates().sort_values('origin'))

# Distribution of cars by origin
plt.figure(figsize=(6, 4))
sns.countplot(data=df, x="origin_label", palette="Set2",
              order=["America", "Europe", "Japan"])
plt.title("Number of Cars by Origin")
plt.xlabel("Origin")
plt.ylabel("Count")
plt.tight_layout()
plt.show()

# Final cleaned dataframe preview
print("\n--- Final cleaned dataframe ---")
df.head()
